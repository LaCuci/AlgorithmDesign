\section{Complexity classes and graph reductions}
\subsection{Exercise: co-NP and its relationship to NP}
Define
\[\mathrm{co\text{-}NP}=\{L\subseteq\bits^*:L^c\in\classNP\}.\]
$\equiv\;L'=L^c\longrightarrow(\equiv\{L'^c\subseteq\bits^*:L'\in\classNP\})$.
\begin{enumerate}
\item Prove that $\classP\subseteq\classNP\cap\mathrm{co\text{-}NP}$.
\end{enumerate}
\sourcefigure[0.72]{assets/08/page-001-p-np-co-np.png}
\begin{enumerate}[start=2]
\item Prove that if $\classNPC\cap\mathrm{co\text{-}NP}\ne\varnothing\Rightarrow\classNP=\mathrm{co\text{-}NP}$.
\end{enumerate}
