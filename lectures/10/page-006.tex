\sourceheading{EXTENSION: Weighted Vertex Cover}
\begin{notesdefinition}
\textbf{DEFINITION} Given a node-weighted undirected graph:
\[\langle G=(V,E),w:V\mapsto\mathbb N^+\rangle\]
(each node is associated to a positive weight), determine a vertex cover $V'\subseteq V$ minimizing
\[C(V')=\sum_{u\in V'}w(u).\]
\end{notesdefinition}
\textbf{OBSERVATION} Letting $w(u)=1$ $\forall u\in V$ we obtain MINIMUM VERTEX COVER.
