\section{Recap: Approximation guarantees and vertex cover}
\begin{itemize}
\item Definition of $\rho(n)$-approximation algorithm $A_\Pi$ for optimization problem $\Pi$:
\[\forall i:|i|=n:\quad\max\left\{\underbrace{\frac{C(A_\Pi(i))}{C(s^*(i))}}_{\mathrm{MIN}},\underbrace{\frac{C(s^*(i))}{C(A_\Pi(i))}}_{\mathrm{MAX}}\right\}\leq\rho(n).\]
\item Definition of PTAS, FPTAS $A_\Pi(i,\varepsilon)$:
\[\rho(n)\leq1+\varepsilon,\qquad T_{A_\Pi}=O(\operatorname{poly}(n)),\qquad T_{A_\Pi}=O\left(\operatorname{poly}\left(n,\frac1\varepsilon\right)\right).\]
\item A maximal matching-based algorithm A-V-C ($G$) with $\rho=2$ (tight) to approximate MINIMUM VERTEX COVER in linear time.
\end{itemize}
