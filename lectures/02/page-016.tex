\begin{notesalgorithm}
\textbf{VERIFY-HAMILTON$(x,y)$}
\begin{notespseudocode}
\pseudocodeline{0}{\textbf{if} $(x\neq\langle G=(V,E)\rangle)$ \textbf{then} \textbf{return} $0$}
\pseudocodeline{0}{\textbf{if} $(y\neq\langle v_1,\ldots,v_{|V|+1}\rangle,\ v_i\in V)$ \textbf{then} \textbf{return} $0$}
\pseudocodeline{0}{\{check that input is ``well formed''\}}
\pseudocodeline{0}{\textbf{if} $(v_1\neq v_{|V|+1})$ \textbf{then} \textbf{return} $0$}
\pseudocodeline{0}{\{candidate cannot be a cycle!\}}
\pseudocodeline{0}{\textbf{if} $(\exists i,j:\ 1\leq i<j\leq|V|:\ v_i=v_j)$ \textbf{then} \textbf{return} $0$}
\pseudocodeline{0}{\{check that $y$ contains all nodes\}}
\pseudocodeline{0}{\textbf{for} $i\leftarrow1$ \textbf{to} $|V|$ \textbf{do}}
\pseudocodeline{1}{\textbf{if} $(\{v_i,v_{i+1}\}\notin E)$ \textbf{then} \textbf{return} $0$}
\pseudocodeline{0}{\{check cycle edges\}}
\pseudocodeline{0}{\textbf{return} $1$}
\end{notespseudocode}
\end{notesalgorithm}

The algorithm is at most quadratic in the size of the input! So we have a polinomial time verification algorithm for the hamiltonian cycle problem. The algorithm runs in polynomial time in the size of the input $x$ and the size of the certificate $y$, so we have that $\mathrm{HAMILTON}\in\classNP$.

\begin{notesdefinition}
\textbf{DEF:} A two-input algorithm $V$ verifies string $x\in\bits^*$ iff $\exists y\in\bits^*:\ V(x,y)=1$.
Any $y:\ V(x,y)=1$ is called a certificate for $x$.
\end{notesdefinition}
