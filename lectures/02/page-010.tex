\sourceheading{FORMAL LANGUAGE REPRESENTATION OF CONCRETE DECISION PROBLEMS}
\textbf{RECAP:} Given an alphabet $\Sigma$, a formal language $L$ over $\Sigma$ is a set of finite strings:
\[
L\subseteq\Sigma^*\quad\text{(Kleene star of }\Sigma\text{)}.
\]
\begin{itemize}
\item $L$ may/may not contain the empty string $\eps$ ($\eps\notin L$ or $\eps\in L$).
\item Empty language: $\varnothing$ ($\{\eps\}\neq\varnothing$!).
\item $L_1\cdot L_2$ concatenation operator:
\[
L_1\cdot L_2=\{z\in\Sigma^*:\exists x\in L_1,y\in L_2:z=\langle x,y\rangle\}.
\]
\item Powers: $L^0=\{\eps\}$; $L^i=L\cdot L^{i-1}$.
\end{itemize}
