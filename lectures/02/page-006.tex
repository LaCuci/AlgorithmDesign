\subsection{Unary Encoding}
If $m\in\mathcal I=\mathbb N-\{0\}$,
\[
e_u(m)=\langle\underbrace{1,\ldots,1}_{m\text{ times}}\rangle\quad\text{(string of }m\text{ 1's)}.
\]
\textbf{NOTE:} $|e_u(m)|=m$.

\textbf{NOTE:}
\[
m=|e_u(m)|=2^{\log_2 m}=\Theta(2^{|e_b(m)|}).
\]

The unary encoding is exponentially longer than the binary encoding!

\subsection{Ripercussions on complexity analysis}
$T(n)$ uses encoding length as the independent variable $n$.

Consider this algorithm:

\begin{notesalgorithm}
\textbf{GAUSS$(m)$}
\begin{notespseudocode}
\pseudocodeline{0}{$\mathrm{sum}\leftarrow0$}
\pseudocodeline{0}{\textbf{for} $i\leftarrow1$ \textbf{to} $m$ \textbf{do} $\mathrm{sum}\leftarrow\mathrm{sum}+i$}
\pseudocodeline{0}{\textbf{return} sum}
\end{notespseudocode}
\end{notesalgorithm}
