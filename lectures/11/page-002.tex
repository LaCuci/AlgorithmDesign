\section{The travelling salesperson problem (TSP)}
\subsection{Decision problem}
Decision version:

\textbf{TSP}
\begin{notesproblem}
&\text{I: }\langle G=(V,E),c,K\rangle:\\
&G\text{ complete, undirected graph}\\
&c:V\times V\to\mathbb N,\text{ symmetric}\\
&\qquad(c(u,v)=c(v,u)),\quad(c(u,u)=0)\\
&K\in\mathbb N-\{0\}\\[4pt]
&\text{Q: Is there a tour (hamiltonian cycle)}\\
&\gamma=\langle v_1,v_2,\ldots,v_{|V|},v_{|V|+1}=v_1\rangle\\
&\qquad(v_i\ne v_j\quad1\leq i<j\leq|V|)\\
&\text{of cost }c(\gamma)=\sum_{i=1}^{|V|}c(v_i,v_{i+1})\leq K\text{?}
\end{notesproblem}
\textbf{OBSERVATION:} There are $|V|!$ tours in $G$.

\subsection{NP-hardness: Reduction from HAMILTON}
We can prove $\mathrm{TSP}\in\classNPH$ by reducing from HAMILTON:

An instance of HAMILTON is an arbitrary undirected graph $G=(V,E)$.

Reduction:
\[\langle G=(V,E)\rangle\overset f\longmapsto\langle G'=(V,E'),c,K\rangle\]
with $E'=\{\{u,v\}:u,v\in V\}$ (complete set),
\[c(u,v)=\begin{cases}1&\text{if }\{u,v\}\in E,\\2&\text{otherwise},\end{cases}\qquad K=|V|.\]
