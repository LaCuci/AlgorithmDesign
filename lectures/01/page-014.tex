Once the length $k^*$ of the shortest path is established, it is easy to determine the actual path through $|E|$ calls to $A_{\mathrm{KUP}}$.

\textbf{Algorithm idea:} progressively eliminate edges which are not necessary to guarantee the existence of a shortest path. To check this, we use $A_{\mathrm{KUP}}$ as an ORACLE.

\begin{notesalgorithm}
\textbf{$A_{\mathrm{SUP}}(\langle G=(V,E),u,z\rangle)$}
\begin{notespseudocode}
\pseudocodeline{0}{$k^*\leftarrow\text{P-LENGTH}(\langle G,u,z\rangle)$ \{see above\}}
\pseudocodeline{0}{\textbf{if} $(k^*=+\infty)$ \textbf{then} \textbf{return} ``no path''}
\pseudocodeline{0}{$*\ E=\{e_1,e_2,\ldots,e_{|E|}\}\ *$}
\pseudocodeline{0}{$E'\leftarrow E$}
\pseudocodeline{0}{\textbf{for} $i\leftarrow1$ \textbf{to} $|E|$ \textbf{do}}
\pseudocodeline{1}{\textbf{if} $A_{\mathrm{KUP}}(\langle G'=(V,E'-\{e_i\}),u,z,k^*\rangle)=\mathrm{yes}$}
\pseudocodeline{2}{\textbf{then} $E'\leftarrow E'-\{e_i\}$}
\pseudocodeline{1}{\{$\exists$ shortest path $\pi^*\in E'$ not using $e_i$\}}
\pseudocodeline{0}{$*\ E'$ contains only the edges of a s.p.!! $*$}
\end{notespseudocode}
