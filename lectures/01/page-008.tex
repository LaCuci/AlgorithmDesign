\subsection{Problem Categories}
\begin{enumerate}
\item Tractable problems: Problems admitting polynomial algorithms
\[
\exists A_\Pi:\quad T_{A_\Pi}(n)=O(n^k),\quad k\geq0.
\]
\item Provably intractable problems : Problems admitting an exponential lower bound
\[
\exists c>1:\ \forall A_\Pi:\quad T_{A_\Pi}(n)=\Omega(c^n).
\]
\item Undecidable /Unsolvable problems : Problems not admitting any algorithm 
\item Intractable problems: problems for which neither an exponential lower bound nor a polynomial upper bound is known
\end{enumerate}

\subsection{HALTING PROBLEM}
\[
\Pi_{\mathrm{HALT}}:\quad
\mathcal I=\{\langle M,x\rangle:\ M\text{ Turing Machine},\ x\in\bits^*\},
\qquad\mathcal S=\{\mathrm{yes},\mathrm{no}\}.
\]
\[
\langle M,x\rangle\mathrel\Pi_{\mathrm{HALT}}\mathrm{yes}
\quad\Longleftrightarrow\quad M(x)\text{ halts}.
\]
