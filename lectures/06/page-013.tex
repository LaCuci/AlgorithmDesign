$\forall1\leq i\leq n$:
\begin{itemize}
\item $\delta_i,\delta'_i$ have a single 1 in the $i$th position of the $n$ most significant digits.
\item $\delta_i$ has 1's in all positions $j$ of the $m$ least significant digits, where $C_j$ contains literal $x_i$.
\item $\delta'_i$ has 1's in all positions $j$ of the $m$ least significant digits, where $C_j$ contains literal $\bar x_i$.
\item All other digits are 0.
\end{itemize}
$\forall1\leq j\leq m$:
\begin{itemize}
\item $s_j$ has a single 1 in the $j$th position of the $m$ least significant ones.
\item $s'_j$ has a single 2 in the $j$th position of the $m$ least significant ones.
\item All other digits are 0.
\end{itemize}
