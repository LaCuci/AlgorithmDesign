\textbf{PROPERTIES OF A FPW}
\begin{enumerate}
\item The first occurrence of each node gives the preorder visit of $T^*$.
\item Each edge of $T^*$ is used 2 times by FPW.
\end{enumerate}
\end{notesresult}
