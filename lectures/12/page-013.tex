We are ready to specify the algorithm:
\begin{notesalgorithm}
\textbf{CHRISTOFIDES ($G=(V,E),c$)}
\begin{notespseudocode}
\pseudocodeline{0}{$T^*=(V,E_{T^*})\gets\mathrm{MST}(G,c)$}
\pseudocodeline{0}{* let $V_{\mathrm{ODD}}^{T^*}$ be the set of nodes of odd degree of $T^*$ and let $E_{\mathrm{ODD}}^{T^*}=\{e=\{u,v\}\in E:u,v\in V_{\mathrm{ODD}}^{T^*}\}$ *}
\pseudocodeline{0}{$M^*\gets\text{M-C-MATCHING}(G'=(V_{\mathrm{ODD}}^{T^*},E_{\mathrm{ODD}}^{T^*}),c)$}
\pseudocodeline{0}{$W\gets\text{EULER-TOUR}(\overline G=(V,E_{T^*}\cup M^*))$}
\pseudocodeline{0}{$\widehat\gamma\gets\mathrm{SHORTCUT}(W)$}
\pseudocodeline{0}{\textbf{return} $\widehat\gamma$}
\end{notespseudocode}
\end{notesalgorithm}
